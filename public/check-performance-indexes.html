Performance Index Checker

Step 1: Check Current Indexes

SELECT
  tablename,
  indexname,
  indexdef
FROM pg_indexes
WHERE tablename IN ('word_progress', 'user_book_preferences')
ORDER BY tablename, indexname;

Step 2: Create Missing Indexes (if needed)

-- word_progress 表的复合索引
CREATE INDEX IF NOT EXISTS idx_word_progress_user_updated
ON word_progress(user_id, updated_at DESC);

CREATE INDEX IF NOT EXISTS idx_word_progress_user_created
ON word_progress(user_id, created_at DESC);

CREATE INDEX IF NOT EXISTS idx_word_progress_user_status
ON word_progress(user_id, status);

-- user_book_preferences 表的复合索引
CREATE INDEX IF NOT EXISTS idx_user_book_prefs_user_accessed
ON user_book_preferences(user_id, last_accessed_at DESC);

Step 3: Verify Index Usage

EXPLAIN ANALYZE
SELECT book_id, status, created_at, updated_at
FROM word_progress
WHERE user_id = 'your-user-id'
ORDER BY updated_at DESC
LIMIT 100;
