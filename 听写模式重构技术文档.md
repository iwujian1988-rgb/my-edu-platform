## 2. 技术方案设计

### 2.1 架构设计原则

遵循 **SOLID原则** 和 **AI_RULES.MD核心原则**：

1. **单一职责 (SRP)**
   - 范围选择逻辑 → 独立的Hook: `useDictationScope`
   - 数据获取逻辑 → 独立的Hook: `useDictationWords`
   - 进度管理逻辑 → 独立的Hook: `useDictationProgress`

2. **开闭原则 (OCP)**
   - 扩展：支持新的scopeType无需修改核心逻辑
   - 封装：API调用封装在Service层

3. **依赖倒置 (DIP)**
   - 依赖抽象接口 (IDictationProgressService)
   - 不依赖具体实现

4. **防御性编程**
   - 所有外部输入必须验证（Zod schema）
   - 所有异步操作必须处理失败（重试机制）
   - 所有边界条件必须考虑（空数据、并发、缓存）

### 3.2 数据迁移策略 [✅ 简化]

由于复用现有表结构，**无需数据迁移**。

**旧进度兼容**：
```typescript
// 旧听写进度存储在resumeState表或localStorage
// 格式：{ mode: 'dictation', bookId, index, totalWords }

// 兼容策略：
// 1. 首次访问时，检查resumeState中的dictation进度
// 2. 如果存在，迁移到user_book_preferences.preferences
// 3. 默认迁移到"全部单词"范围 (scopeType = 'all')

async function migrateOldDictationProgress(userId: string, bookId: string) {
  const supabase = await createClient()

  // 1. 从resumeState读取旧进度
  const { data: oldProgress } = await supabase
    .from('resume_state')
    .select('*')
    .eq('user_id', userId)
    .eq('mode', 'dictation')
    .eq('book_id', bookId)
    .single()

  if (!oldProgress) return  // 无旧进度，直接返回

  // 2. 转换为新格式
  const progressKey = `dictation_progress_${bookId}_all`
  const newProgressData = {
    currentIndex: oldProgress.index || 0,
    totalWords: oldProgress.total_words || 0,
    lastStudyTime: oldProgress.last_study_time || Date.now(),
    scopeType: 'all'
  }

  // 3. 保存到user_book_preferences
  const { data: existing } = await supabase
    .from('user_book_preferences')
    .select('preferences')
    .eq('user_id', userId)
    .eq('book_id', bookId)
    .single()

  const currentPreferences = existing?.preferences || {}
  const updatedPreferences = {
    ...currentPreferences,
    [progressKey]: newProgressData
  }

  await supabase
    .from('user_book_preferences')
    .upsert({
      user_id: userId,
      book_id: bookId,
      preferences: updatedPreferences
    }, {
      onConflict: 'user_id,book_id'
    })

  console.log(`✅ 旧进度已迁移: ${bookId}`)
}
```

**实现要点** [✅ 修改：完善Zod验证]：
```typescript
// /api/words/stats/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { createRouteHandlerClient } from '@supabase/auth-helpers-nextjs'
import { cookies } from 'next/headers'
import { z } from 'zod'
import { fromZodError } from 'zod-validation-error'  // ✅ 新增

// ✅ 完善的Zod Schema
const GetStatsSchema = z.object({
  bookId: z.string()
    .min(1, 'bookId不能为空')
    .uuid('bookId格式错误')  // ✅ 新增UUID验证
})

export async function GET(request: NextRequest) {
  try {
    // 1. 参数验证 (使用parse而不是safeParse)
    const searchParams = Object.fromEntries(request.nextUrl.searchParams)

    // ✅ 使用parse，自动抛出详细错误
    const { bookId } = GetStatsSchema.parse(searchParams)

    // 2. 初始化数据库连接
    const supabase = createRouteHandlerClient({ cookies })

    // 3. 获取当前用户
    const { data: { user }, error: authError } = await supabase.auth.getUser()
    if (authError || !user) {
      return NextResponse.json({
        success: false,
        error: '未授权访问',
        code: 'UNAUTHORIZED'
      }, { status: 401 })
    }

    // 4. ✅ 尝试从缓存获取（新增）
    const cachedStats = await cacheService.getStats(user.id, bookId)
    if (cachedStats) {
      return NextResponse.json({
        success: true,
        data: cachedStats,
        _cached: true  // 标记来自缓存
      })
    }

    // 5. 查询统计数据 (使用聚合查询，性能优化)
    const { data: stats, error: statsError } = await supabase
      .from('word_progress')
      .select('status', { count: 'exact' })
      .eq('user_id', user.id)
      .eq('book_id', bookId)

    if (statsError) {
      console.error('❌ [Stats API] 查询失败:', statsError)
      throw statsError
    }

    // 6. 获取总词数
    const { count: totalCount, error: countError } = await supabase
      .from('words')
      .select('*', { count: 'exact', head: true })
      .eq('book_id', bookId)

    if (countError) {
      console.error('❌ [Stats API] 获取总词数失败:', countError)
      throw countError
    }

    // 7. 聚合数据
    const statusCount = {
      all: totalCount || 0,
      unknown: 0,
      fuzzy: 0,
      known: 0,
      new: totalCount || 0 // 默认都是未标注
    }

    if (stats) {
      stats.forEach(item => {
        if (item.status in statusCount) {
          statusCount[item.status] = (statusCount[item.status] || 0) + 1
          statusCount.new-- // 减少未标注数量
        }
      })
    }

    // 8. ✅ 写入缓存（新增）
    await cacheService.setStats(user.id, bookId, statusCount)

    return NextResponse.json({
      success: true,
      data: statusCount
    })

  } catch (error) {
    // ✅ 完善的错误处理
    if (error instanceof z.ZodError) {
      // 返回所有验证错误
      return NextResponse.json({
        success: false,
        error: fromZodError(error).message,  // ✅ 格式化所有错误
        code: 'INVALID_PARAMS',
        details: error.errors  // ✅ 返回详细错误列表
      }, { status: 400 })
    }

    console.error('❌ [Stats API] 服务器错误:', error)
    return NextResponse.json({
      success: false,
      error: '服务器内部错误',
      code: 'INTERNAL_ERROR'
    }, { status: 500 })
  }
}
```

**实现要点** [✅ 修改：扩展mode参数]：
```typescript
// /api/flashcard-progress/route.ts (修改现有文件)

import { createClient, getCurrentUser } from '@/lib/supabase/server'
import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'
import { fromZodError } from 'zod-validation-error'

// ✅ Zod Schema: 支持mode参数
const GetProgressSchema = z.object({
  bookId: z.string().min(1).uuid('bookId格式错误'),
  scopeType: z.enum(['all', 'unknown', 'fuzzy', 'known', 'new']),
  mode: z.enum(['flashcards', 'dictation']).default('flashcards')  // ✅ 新增
})

const PostProgressSchema = z.object({
  bookId: z.string().min(1).uuid('bookId格式错误'),
  scopeType: z.enum(['all', 'unknown', 'fuzzy', 'known', 'new']),
  mode: z.enum(['flashcards', 'dictation']).default('flashcards'),  // ✅ 新增
  currentIndex: z.number().int().min(0, 'currentIndex不能为负数'),
  totalWords: z.number().int().min(0, 'totalWords不能为负数')
})

/**
 * GET /api/flashcard-progress?bookId=xxx&scopeType=xxx&mode=flashcards|dictation
 * 获取指定范围和模式的学习进度
 */
export async function GET(request: NextRequest) {
  const user = await getCurrentUser()

  if (!user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  try {
    const searchParams = Object.fromEntries(request.nextUrl.searchParams)

    // ✅ 使用Zod验证
    const { bookId, scopeType, mode } = GetProgressSchema.parse(searchParams)

    const supabase = await createClient()

    // 从 user_book_preferences 表获取进度
    const { data: preferences, error } = await supabase
      .from('user_book_preferences')
      .select('preferences')
      .eq('user_id', user.id)
      .eq('book_id', bookId)
      .maybeSingle()

    if (error || !preferences) {
      return NextResponse.json({
        success: true,
        data: null
      })
    }

    const allPreferences = (preferences as any)?.preferences || {}
```

```typescript
// lib/cache/redis.ts

import { Redis } from 'ioredis'

interface CacheStats {
  hits: number
  misses: number
  errors: number
}

class CacheService {
  private redis: Redis
  private readonly DEFAULT_TTL = 300  // 5分钟
  private readonly LOCK_TTL = 10      // 锁的TTL：10秒
  private stats: CacheStats = { hits: 0, misses: 0, errors: 0 }

  constructor() {
    const redisUrl = process.env.REDIS_URL || 'redis://localhost:6379'
    this.redis = new Redis(redisUrl, {
      retryDelayOnFailover: 100,
      enableReadyCheck: true,
      maxRetriesPerRequest: 3
    })

    this.redis.on('error', (error) => {
      console.error('❌ Redis error:', error)
      this.stats.errors++
    })
  }

  /**
   * 获取统计数据（带缓存击穿保护）
   */
  async getStats(userId: string, bookId: string): Promise<DictationStats | null> {
    const cacheKey = `stats:${userId}:${bookId}`

    try {
      // 1. 尝试从缓存读取
      const cached = await this.redis.get(cacheKey)
      if (cached) {
        this.stats.hits++
        return JSON.parse(cached)
      }

      this.stats.misses++
      return null  // 缓存未命中
    } catch (error) {
      console.error('❌ Cache get error:', error)
      this.stats.errors++
      return null  // 缓存失败，降级到数据库查询
    }
  }

  /**
   * 设置统计数据（带随机TTL，避免雪崩）
   */
  async setStats(userId: string, bookId: string, stats: DictationStats): Promise<void> {
    const cacheKey = `stats:${userId}:${bookId}`

    try {
      // 添加随机过期时间，避免雪崩
      const randomTTL = this.DEFAULT_TTL + Math.floor(Math.random() * 60)  // 300-360秒

      await this.redis.setex(cacheKey, randomTTL, JSON.stringify(stats))
    } catch (error) {
      console.error('❌ Cache set error:', error)
      this.stats.errors++
      // 静默失败，不影响业务逻辑
    }
  }

  /**
   * 失效缓存（当word_progress更新时调用）
   */
  async invalidateStats(userId: string, bookId: string): Promise<void> {
    const cacheKey = `stats:${userId}:${bookId}`

    try {
      await this.redis.del(cacheKey)
      console.log(`✅ Cache invalidated: ${cacheKey}`)
    } catch (error) {
      console.error('❌ Cache invalidate error:', error)
      this.stats.errors++
    }
  }

  /**
   * 获取缓存统计信息
   */
  getStats(): CacheStats {
    return { ...this.stats }
  }

  /**
   * 关闭连接
   */
  async close(): Promise<void> {
    await this.redis.quit()
  }
}

export const cacheService = new CacheService()
```

### 6.4 重试工具实现 [新增]

```typescript
// lib/retry/withRetry.ts

import retry from 'async-retry'

interface RetryOptions {
  maxAttempts: number
  baseDelay: number
  context?: string
}

/**
 * 带重试的异步函数包装器
 *
 * @param fn - 要执行的异步函数
 * @param options - 重试配置
 * @returns 函数执行结果
 */
export async function withRetry<T>(
  fn: () => Promise<T>,
  options: RetryOptions
): Promise<T> {
  const { maxAttempts, baseDelay, context } = options

  return retry(
    async (bail: (err: Error) => void) => {
      try {
        return await fn()
      } catch (error) {
        // 如果是AbortError，不重试
        if (error instanceof Error && error.name === 'AbortError') {
          bail(error)
          throw error  // TypeScript需要
        }

        throw error
      }
    },
    {
      retries: maxAttempts - 1,
      factor: 2,  // 指数退避
      minTimeout: baseDelay,
      maxTimeout: baseDelay * 4,
      onRetry: (error, attempt) => {
        console.warn(
          `⚠️ [${context || 'withRetry'}] 第${attempt}次重试...`,
          error.message
        )
      }
    }
  )
}
```

```typescript
// services/progressManager.ts

import { dictationService } from './dictationService'
import { DictationScopeType } from '@/types/dictation'

/**
 * 听写进度管理器
 *
 * 职责：
 * 1. 防抖保存（避免频繁API调用）
 * 2. 批量保存（累积多个进度一起保存）
 * 3. 关键场景立即保存（页面卸载等）
 */
class DictationProgressManager {
  private pendingSaves = new Map<string, number>()
  private saveTimer: NodeJS.Timeout | null = null
  private readonly DEBOUNCE_DELAY = 1000  // 1秒防抖

  /**
   * 保存进度（防抖）
   */
  saveProgress(
    bookId: string,
    scopeType: DictationScopeType,
    currentIndex: number,
    totalWords: number
  ): void {
    const key = `${bookId}:${scopeType}`

    // 1. 累积待保存的进度
    this.pendingSaves.set(key, currentIndex)

    // 2. 清除之前的定时器
    if (this.saveTimer) {
      clearTimeout(this.saveTimer)
    }

    // 3. 设置新的防抖定时器
    this.saveTimer = setTimeout(async () => {
      await this.flush()
    }, this.DEBOUNCE_DELAY)
  }

  /**
   * 立即保存所有待保存的进度（用于页面卸载等关键场景）
   */
  async flush(): Promise<void> {
    if (this.saveTimer) {
      clearTimeout(this.saveTimer)
      this.saveTimer = null
    }

    if (this.pendingSaves.size === 0) {
      return
    }

    console.log(`💾 [ProgressManager] 批量保存 ${this.pendingSaves.size} 个进度`)

    // 批量保存所有待保存的进度
    const savePromises = Array.from(this.pendingSaves.entries()).map(
      async ([key, index]) => {
        const [bookId, scopeType] = key.split(':')
        const totalWords = 0  // TODO: 从某处获取totalWords

        try {
          await dictationService.saveProgress(bookId, scopeType, index, totalWords)
          console.log(`✅ 保存成功: ${key} -> ${index}`)
        } catch (error) {
          console.error(`❌ 保存失败: ${key}`, error)
        }
      }
    )

    await Promise.all(savePromises)

    // 清空待保存队列
    this.pendingSaves.clear()
  }

  /**
   * 获取待保存的进度数量
   */
  getPendingCount(): number {
    return this.pendingSaves.size
  }
}
```

```typescript
      if (!bookId || !scopeType) return

      setLoading(true)
      setError(null)

      try {
        const data = await dictationService.getProgress(bookId, scopeType)

        if (!mounted) return

        if (data) {
          setProgress(data)
        } else {
          // 没有进度，使用默认值
          setProgress({
            currentIndex: 0,
            totalWords: 0,
            lastStudyTime: Date.now()
          })
        }
      } catch (err) {
        console.error('❌ [useDictationProgress] 获取进度失败:', err)
        if (!mounted) return
        setError(err as Error)
      } finally {
        if (mounted) {
          setLoading(false)
        }
      }
    }

    fetchProgress()

    return () => {
      mounted = false
    }
  }, [bookId, scopeType])

  /**
   * 保存进度（使用progressManager防抖）
   */
  const saveProgress = useCallback((currentIndex: number) => {
    try {
      // 1. ✅ 使用progressManager防抖保存
      progressManager.saveProgress(bookId, scopeType, currentIndex, totalWords)

      // 2. 保存到resumeState（用于首页"继续学习"）
      saveResumeState(bookId, 'dictation', {
        scope: scopeType,
        index: currentIndex,
        totalWords: totalWords
      })

      // 3. 更新本地状态
      setProgress(prev => ({
        ...prev,
        currentIndex,
        lastStudyTime: Date.now()
      }))

      console.log(`✅ [useDictationProgress] 进度已加入保存队列: ${currentIndex + 1}/${totalWords}`)
    } catch (err) {
      console.error('❌ [useDictationProgress] 保存进度失败:', err)
      // 静默失败，不影响用户继续学习
    }
  }, [bookId, scopeType, totalWords])

  // ✅ 组件卸载时，确保保存所有待保存的进度
  useEffect(() => {
    return () => {
      progressManager.flush()
    }
  }, [])

  return {
    progress,
    loading,
    error,
    saveProgress
  }
}
```

## 7. 并发控制设计 [新增]

### 7.1 状态机管理

```typescript
// hooks/useDictationPageState.ts

type PageState = 'idle' | 'loading' | 'saving' | 'switching'

/**
 * 听写页面状态管理
 *
 * 职责：
 * 1. 防止并发操作冲突
 * 2. 提供操作锁机制
 * 3. 确保状态一致性
 */
export function useDictationPageState() {
  const [pageState, setPageState] = useState<PageState>('idle')
  const [currentOperation, setCurrentOperation] = useState<string | null>(null)

  /**
   * 执行操作（带状态检查）
   */
  async function executeOperation<T>(
    operationName: string,
    state: PageState,
    fn: () => Promise<T>
  ): Promise<T> {
    // 1. 检查是否可以执行操作
    if (pageState === 'saving' || pageState === 'switching') {
      throw new Error(`无法执行${operationName}：当前正在${currentOperation}`)
    }

    // 2. 设置状态
    setPageState(state)
    setCurrentOperation(operationName)

    try {
      // 3. 执行操作
      const result = await fn()
      return result
    } finally {
      // 4. 恢复状态
      setPageState('idle')
      setCurrentOperation(null)
    }
  }

  return {
    pageState,
    currentOperation,
    isIdle: pageState === 'idle',
    isLoading: pageState === 'loading',
    isSaving: pageState === 'saving',
    isSwitching: pageState === 'switching',
    canOperate: pageState === 'idle',
    executeOperation
  }
}
```

### 7.2 使用示例

```typescript
// app/study/[bookId]/dictation/page.tsx

export default function DictationPage() {
  // ... 其他hooks ...

  const { pageState, canOperate, executeOperation } = useDictationPageState()

  /**
   * 切换到下一个单词（带并发控制）
   */
  const handleNext = async () => {
    if (!canOperate) {
      console.warn('⚠️ 无法切题：正在保存中或切换中')
      return
    }

    await executeOperation(
      '切题',
      'saving',
      async () => {
        // 1. 保存进度
        await saveProgress(currentIndex)

        // 2. 切换到下一个单词
        const nextIndex = currentIndex + 1
        setCurrentIndex(nextIndex)
      }
    )
  }

  /**
   * 切换范围（带并发控制）
   */
  const handleScopeChange = async (newScope: string) => {
    if (!canOperate) {
      console.warn('⚠️ 无法切换范围：正在保存中')
      return
    }

    await executeOperation(
      '切换范围',
      'switching',
      async () => {
        // 1. 取消当前请求
        abortCurrentRequest()

        // 2. 切换范围
        await switchScope(newScope)
      }
    )
  }

  return (
    <div>
      {/* 禁用按钮样式 */}
      <button
        onClick={handleNext}
        disabled={!canOperate}
        className={canOperate ? 'btn-primary' : 'btn-disabled'}
      >
        {pageState === 'saving' ? '保存中...' : '下一个'}
      </button>
    </div>
  )
}
```
